			limit = {

	type = character_event
	title = votc_placeholder
	theme = clipboard_transfer_event_theme
	desc = talk_event.9001.desc
	window = fullscreen_event

		scope:recipient = {
			log_gamedata_v3s = yes
		}
		# VOTC-182: AI-initiated conversations. This marker tells the app that
		# the initiating AI character (votc_ai_initiator) speaks first. The line
		# is additive and guarded, so player-initiated conversations are unchanged.
		if = {
			limit = { has_global_variable = votc_ai_initiator }
			debug_log = "VOTC:IN/;/ai_speaks_first/;/[GetGlobalVariable('votc_ai_initiator').GetCharacter.GetID]/;/"
		}
	}

	#this widget is what calls the EventWindowData.CopyToClipboard guifunction,
	widget = {
        gui = "event_window_widget_talk_v2"
        container = "custom_widgets_container"
    }
}

#VOTC-182: AI-initiated conversation request. Fired by votc_ai_initiate_conversation_effect
#(yearly_playable_pulse -> votc_ai_initiate_conversation_pulse) when an eligible AI character is
#at the player's location. The AI already picked the initiator (scope:recipient), the extra
#participants (mcc_characters_selected: 75% none, 25% 2-15) and the scene (talk_scene). Accept starts the conversation
#through the normal flow (mcc_event_v2.8999), where the initiating AI character speaks first (see
#the ai_speaks_first marker in 8999's immediate). Decline cleans up the prepared state.
mcc_event_v2.1010 = {
	type = character_event
	theme = diplomacy_family_focus
	title = votc_ai_initiate_event_title
	desc = votc_ai_initiate_event_desc
	window = character_event
	left_portrait = {
		character = scope:recipient
	}

	option = {
		name = votc_ai_initiate_accept
		# VOTC-182 4.8: open the setup popup instead of starting directly.
		trigger_event = mcc_event_v2.1011
	}

	option = {
		name = votc_ai_initiate_decline
		# VOTC-182 4.6/4.12: -5 opinion from the AI initiator toward the player.
		# Only applies to AI-initiated conversations.
		if = {
			limit = { has_global_variable = votc_ai_initiator }
			scope:recipient = {
				add_opinion = {
					target = root
					modifier = conversation_opinion
					opinion = -5
				}
			}
		}
		# Cleanup: nothing was started, strip the prepared state.
		remove_global_variable = votc_ai_initiator
		remove_global_variable = talk_scene
		clear_variable_list = mcc_characters_selected
	}
}
