#==========================================================================
#  VOTC-194: NPC Initiative - events
#  namespace = votc_ce_initiative
#
#   .1     hidden  tick loop (root = player)
#   .2     prompt  "<NPC> wants to talk to you"  -> accept / decline
#   .3     hidden  commit an accepted invitation (open the CE talk flow)
#   .4     hidden  arrival of a travelling NPC -> back to .3
#   .9     hidden  the player declined
#   .5     hidden  letter heartbeat (a FIXED cadence, no dice)
#   .152   hidden  a letter arrived through run/votc_ce_npcltr.txt
#   .153   window  the letter itself (+ write-back)
#   .160   token   invisible clipboard token for a letter
#   .100   settings hub
#   .101   toggle on / off
#   .110   interval page      .111 / .112 / .113 / .114
#   .120   probability page   .121 / .122 / .123
#   .130   test now (talk)
#   .140   letters on / off
#   .170   letter interval page   .171 / .172 / .173
#   .174   write one right now
#   .180   incoming-letters page (frequency + toggle + clear + details)
#   .181-.184 hidden  incoming-letter frequency setters
#   .185   hidden  incoming letters on / off (VOTC-194 14.3.1)
#   .186   hidden  clear every pending incoming-letter slot (VOTC-194 14.3.2)
#   .190   AI-conversation presets page (VOTC-194 14.2.1)
#   .191-.194 hidden  AI-conversation preset setters (VOTC-194 14.2.2)
#==========================================================================

namespace = votc_ce_initiative

#--------------------------------------------------------------------------
#  .1  Hidden heartbeat.  Re-arms itself every <votc_ce_initiative_freq>
#      days and rolls for a random already-known NPC.
#--------------------------------------------------------------------------
votc_ce_initiative.1 = {
	type = character_event
	hidden = yes
	orphan = yes

	immediate = {
		# 400 days: the longest interval is a full year, and a shorter life made
		# the bootstrap watchdog fire an out-of-schedule round every year.
		set_global_variable = {
			name = votc_ce_initiative_heartbeat
			value = root
			days = 400
		}
		if = {
			limit = {
				is_ai = no
				is_alive = yes
				NOT = { has_global_variable = votc_ce_initiative_off }
			}
			votc_ce_initiative_try = yes
		}
		# VOTC-194 13.1.2: refresh the rolling quarter for the shared letter
		# budget.  Re-seeding the counter to 0 with a 91-day expiry gives a true
		# quarterly reset no matter which cadence the letters run on.
		set_global_variable = {
			name = votc_letters_this_quarter
			value = 0
			days = 91
		}
		# Re-arm.
		if = {
			limit = { global_var:votc_ce_initiative_freq = 30 }
			trigger_event = {
				id = votc_ce_initiative.1
				days = 30
			}
		}
		else_if = {
			limit = { global_var:votc_ce_initiative_freq = 91 }
			trigger_event = {
				id = votc_ce_initiative.1
				days = 91
			}
		}
		else_if = {
			limit = { global_var:votc_ce_initiative_freq = 182 }
			trigger_event = {
				id = votc_ce_initiative.1
				days = 182
			}
		}
		else = {
			trigger_event = {
				id = votc_ce_initiative.1
				days = 365
			}
		}
	}
}

#--------------------------------------------------------------------------
#  .5  Letter heartbeat.  A FIXED cadence - no dice.  Re-arms itself every
#      <votc_ce_initiative_letter_freq> days and writes one letter from a
#      weighted-eligible target.  votc_ce_initiative_letter_pending keeps a
#      second letter from stacking on top of one still being generated.
#--------------------------------------------------------------------------
votc_ce_initiative.5 = {
	type = character_event
	hidden = yes
	orphan = yes

	immediate = {
		set_global_variable = {
			name = votc_ce_initiative_letter_heartbeat
			value = root
			days = 400
		}
		if = {
			limit = {
				is_ai = no
				is_alive = yes
				NOT = { has_global_variable = votc_ce_initiative_off }
				NOT = { has_global_variable = votc_ce_initiative_no_letters }
				NOT = { exists = global_var:votc_ce_initiative_letter_pending }
			}
			votc_ce_initiative_letter_pick = yes
		}
		if = {
			limit = { global_var:votc_ce_initiative_letter_freq = 90 }
			trigger_event = { id = votc_ce_initiative.5 days = 90 }
		}
		else_if = {
			limit = { global_var:votc_ce_initiative_letter_freq = 182 }
			trigger_event = { id = votc_ce_initiative.5 days = 182 }
		}
		else = {
			trigger_event = { id = votc_ce_initiative.5 days = 365 }
		}
	}
}

#--------------------------------------------------------------------------
#  .2  The actual "someone wants to talk to you" prompt.
#--------------------------------------------------------------------------
votc_ce_initiative.2 = {
	type = character_event
	theme = diplomacy_family_focus
	title = votc_ce_initiative.2.title
	desc = votc_ce_initiative.2.desc
	window = character_event

	left_portrait = {
		character = global_var:votc_ce_initiative_char
	}

	immediate = {
		global_var:votc_ce_initiative_char = {
			save_scope_as = recipient
		}
	}

	# Already here -> jump straight into the CE conversation setup.
	option = {
		name = votc_ce_initiative_accept_here
		trigger = {
			exists = global_var:votc_ce_initiative_char
			global_var:votc_ce_initiative_char = {
				is_at_same_location = root
			}
		}
		trigger_event = votc_ce_initiative.3
	}

	# Somewhere else -> they will come to you first.
	option = {
		name = votc_ce_initiative_accept_travel
		trigger = {
			NOT = {
				global_var:votc_ce_initiative_char = {
					is_at_same_location = root
				}
			}
		}
		trigger_event = votc_ce_initiative.3
	}

	option = {
		name = votc_ce_initiative_decline
		trigger_event = votc_ce_initiative.9
	}
}

#--------------------------------------------------------------------------
#  .3  Accepted.  Either open the CE talk flow right away, or send the NPC
#      travelling towards the player.
#
#  We reuse the VOTC-182 conversation pipeline: arm votc_ai_initiator /
#  talk_scene / mcc_characters_selected, then trigger mcc_event_v2.1010 so
#  the player gets the normal scene/participant setup with AI speaking first.
#--------------------------------------------------------------------------
votc_ce_initiative.3 = {
	type = character_event
	hidden = yes
	orphan = yes

	immediate = {
		# root is the player here (this event always runs on the player), so
		# record them as the counterparty the CE talk flow must use.
		set_global_variable = {
			name = votc_ce_initiative_player
			value = root
			days = 3
		}
		if = {
			limit = { exists = global_var:votc_ce_initiative_char }
			global_var:votc_ce_initiative_char = {
				save_scope_as = recipient
			}
			if = {
				limit = {
					scope:recipient = { is_at_same_location = root }
				}
				# Hand over to the CE conversation flow.  votc_ai_initiator tells
				# the pipeline which AI character speaks first; the scene and
				# participants are prepared the same way the old ai-initiate did.
				set_global_variable = {
					name = votc_ai_initiator
					value = scope:recipient
				}
				votc_build_talk_scene_flags = yes
				random_in_list = {
					variable = talk_scene_flags
					set_global_variable = {
						name = talk_scene
						value = this
					}
				}
				clear_variable_list = talk_scene_flags
				root = {
					trigger_event = mcc_event_v2.1010
				}
			}
			else_if = {
				limit = {
					scope:recipient = { is_travelling = no }
				}
				scope:recipient = {
					start_travel_plan = {
						destination = root.location
						on_arrival_event = votc_ce_initiative.4
					}
				}
				set_global_variable = {
					name = votc_ce_initiative_player
					value = root
					days = 730
				}
			}
			else = {
				# Fallback: the NPC was already on the road when the player
				# accepted.  Open the conversation anyway instead of stalling.
				set_global_variable = {
					name = votc_ai_initiator
					value = scope:recipient
				}
				votc_build_talk_scene_flags = yes
				random_in_list = {
					variable = talk_scene_flags
					set_global_variable = {
						name = talk_scene
						value = this
					}
				}
				clear_variable_list = talk_scene_flags
				root = {
					trigger_event = mcc_event_v2.1010
				}
			}
		}
	}
}

#--------------------------------------------------------------------------

#--------------------------------------------------------------------------
#  .153  The letter itself.
#
#  The stock letter parchment is deliberately NOT reused: it renders a fixed
#  localization key around votc_latest_letter.  Here the heading is raw
#  localization text (which IS rich-text parsed, so #T really renders large)
#  and the body comes out of the reply artifact as a data function (which is
#  NOT parsed - plain text with real newlines, exactly what the model wrote).
#--------------------------------------------------------------------------
votc_ce_initiative.153 = {
	type = character_event
	theme = diplomacy_family_focus
	title = votc_ce_initiative.153.title
	desc = votc_ce_initiative.153.desc
	window = character_event

	left_portrait = {
		# NOT scope:actor - a plain character_event carries no actor scope.  The
		# sender travels in a global variable so this also works when the letter
		# arrives long after the request (the round trip is tens of game days).
		character = global_var:votc_ce_initiative_letter_char
	}

	immediate = {
		# Intentionally empty.  The window is rendered AFTER this immediate
		# runs, and it reads both votc_ce_initiative_letter_char and the reply
		# artifact - clearing either here would blank the letter.
	}

	# Write-back: open the existing write-message flow with the sender
	# preselected (message_event.360 reads scope:recipient as the addressee).
	option = {
		name = votc_ce_initiative_letter_write_back
		trigger = {
			exists = global_var:votc_ce_initiative_letter_char
			global_var:votc_ce_initiative_letter_char = { is_alive = yes }
		}
		if = {
			limit = { exists = global_var:votc_ce_initiative_letter_char }
			global_var:votc_ce_initiative_letter_char = {
				save_scope_as = recipient
			}
			scope:recipient = {
				save_scope_as = actor
			}
		}
		# The reply is over: release the guard and drop the throw-away journal.
		votc_ce_initiative_letter_destroy = yes
		votc_ce_initiative_letter_cleanup = yes
		trigger_event = message_event.360
	}

	option = {
		name = votc_ce_initiative_letter_confirm
	}

	after = {
		# The letter has been read: drop the journal and release the
		# one-letter-per-interval gate so the next interval can write again.
		votc_ce_initiative_letter_destroy = yes
		votc_ce_initiative_letter_cleanup = yes
	}
}

#  .101  Toggle the whole feature
#--------------------------------------------------------------------------
votc_ce_initiative.101 = {
	type = character_event
	hidden = yes
	orphan = yes

	immediate = {
		if = {
			limit = { has_global_variable = votc_ce_initiative_off }
			remove_global_variable = votc_ce_initiative_off

				limit = { NOT = { exists = global_var:votc_ce_initiative_heartbeat } }

			}

		else = {

				name = votc_ce_initiative_off
				value = yes
			}
			if = {
				limit = { exists = global_var:votc_ce_initiative_player }
				remove_global_variable = votc_ce_initiative_player
			}
		}
		trigger_event = votc_ce_initiative.100
	}
}

#--------------------------------------------------------------------------
#  .185  Incoming letters on / off (VOTC-194 14.3.1).  Replaces the old
#        votc_enable/disable_incoming_letters_decision pair.  The two hub
#        options are state-named with inverted gating (see .180), so this
#        event only has to flip the state - it never needs to know which
#        label was clicked.
#--------------------------------------------------------------------------
votc_ce_initiative.185 = {
	type = character_event
	hidden = yes
	orphan = yes

	immediate = {
		if = {
			limit = { has_variable = votc_incoming_letters_enabled }
			remove_variable = votc_incoming_letters_enabled
			# Opt-out marker: an explicit "off" must never be silently reverted by
			# the game-start default-on hook (mirrors the old disable decision).
			set_variable = {
				name = votc_incoming_letters_opt_out
				value = 1
			}
		}
		else = {
			# Default-on policy: shared with the game-start hook and the enable
			# scripted gui so all three paths set identical defaults.
			votc_incoming_letters_apply_default_settings_effect = yes
		}
		trigger_event = votc_ce_initiative.180
	}
}
