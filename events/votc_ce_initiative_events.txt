#--------------------------------------------------------------------------
#  .1  Hidden heartbeat.  Re-arms itself every <votc_ce_initiative_freq>
#      days and rolls for a random already-known NPC.
#--------------------------------------------------------------------------
votc_ce_initiative.1 = {
	type = character_event
	hidden = yes
	orphan = yes

	immediate = {
		# 400 days: the longest interval is a full year, and a shorter life made
		# the bootstrap watchdog fire an out-of-schedule round every year.
		set_global_variable = {
			name = votc_ce_initiative_heartbeat
			value = root
			days = 400
		}
		if = {
			limit = {
				is_ai = no
				is_alive = yes
				NOT = { has_global_variable = votc_ce_initiative_off }
			}
			votc_ce_initiative_try = yes
		}
		# VOTC-194 13.1.2: refresh the rolling quarter for the shared letter
		# budget.  Re-seeding the counter to 0 with a 91-day expiry gives a true
		# quarterly reset no matter which cadence the letters run on.
		set_global_variable = {
			name = votc_letters_this_quarter
			value = 0
			days = 91
		}
		# Re-arm.
		if = {
			limit = { global_var:votc_ce_initiative_freq = 30 }
			trigger_event = {
				id = votc_ce_initiative.1
				days = 30
			}
		}
		else_if = {
			limit = { global_var:votc_ce_initiative_freq = 91 }
			trigger_event = {
				id = votc_ce_initiative.1
				days = 91
			}
		}
		else_if = {
			limit = { global_var:votc_ce_initiative_freq = 182 }
			trigger_event = {
				id = votc_ce_initiative.1
				days = 182
			}
		}
		else = {
			trigger_event = {
				id = votc_ce_initiative.1
				days = 365
			}
		}
	}
}

#--------------------------------------------------------------------------
#  .5  Letter heartbeat.  A FIXED cadence - no dice.  Re-arms itself every
#      <votc_ce_initiative_letter_freq> days and writes one letter from a
#      weighted-eligible target.  votc_ce_initiative_letter_pending keeps a
#      second letter from stacking on top of one still being generated.
#--------------------------------------------------------------------------
votc_ce_initiative.5 = {
	type = character_event
	hidden = yes
	orphan = yes

	immediate = {
		set_global_variable = {
			name = votc_ce_initiative_letter_heartbeat
			value = root
			days = 400
		}
		if = {
			limit = {
				is_ai = no
				is_alive = yes
				NOT = { has_global_variable = votc_ce_initiative_off }
				NOT = { has_global_variable = votc_ce_initiative_no_letters }
				NOT = { exists = global_var:votc_ce_initiative_letter_pending }
			}
			votc_ce_initiative_letter_pick = yes
		}
		if = {
			limit = { global_var:votc_ce_initiative_letter_freq = 90 }
			trigger_event = { id = votc_ce_initiative.5 days = 90 }
		}
		else_if = {
			limit = { global_var:votc_ce_initiative_letter_freq = 182 }
			trigger_event = { id = votc_ce_initiative.5 days = 182 }
		}
		else_if = {
			limit = { global_var:votc_ce_initiative_letter_freq = 45 }
			trigger_event = { id = votc_ce_initiative.5 days = 45 }
		}
		else = {
			trigger_event = { id = votc_ce_initiative.5 days = 365 }
		}
	}
}

#--------------------------------------------------------------------------
#  .3  Accepted.  Either open the CE talk flow right away, or send the NPC
#      travelling towards the player.
#
#  We reuse the VOTC-182 conversation pipeline: arm votc_ai_initiator /
#  talk_scene / mcc_characters_selected, then trigger mcc_event_v2.1010 so
#  the player gets the normal scene/participant setup with AI speaking first.
#--------------------------------------------------------------------------
votc_ce_initiative.3 = {
	type = character_event
	hidden = yes
	orphan = yes

	immediate = {
		# root is the player here (this event always runs on the player), so
		# record them as the counterparty the CE talk flow must use.
		set_global_variable = {
			name = votc_ce_initiative_player
			value = root
			days = 3
		}
		if = {
			limit = { exists = global_var:votc_ce_initiative_char }
			global_var:votc_ce_initiative_char = {
				save_scope_as = recipient
			}
			if = {
				limit = {
					scope:recipient = { is_at_same_location = root }
				}
				# Hand over to the CE conversation flow.  votc_ai_initiator tells
				# the pipeline which AI character speaks first; the scene and
				# participants are prepared the same way the old ai-initiate did.
				set_global_variable = {
					name = votc_ai_initiator
					value = scope:recipient
				}
				votc_build_talk_scene_flags = yes
				random_in_list = {
					variable = talk_scene_flags
					set_global_variable = {
						name = talk_scene
						value = this
					}
				}
				clear_variable_list = talk_scene_flags
				root = {
					trigger_event = mcc_event_v2.1010
				}
			}
			else_if = {
				limit = {
					scope:recipient = { is_travelling = no }
				}
				scope:recipient = {
					start_travel_plan = {
						destination = root.location
						on_arrival_event = votc_ce_initiative.4
					}
				}
				set_global_variable = {
					name = votc_ce_initiative_player
					value = root
					days = 730
				}
			}
			else = {
				# Fallback: the NPC was already on the road when the player
				# accepted.  Open the conversation anyway instead of stalling.
				set_global_variable = {
					name = votc_ai_initiator
					value = scope:recipient
				}
				votc_build_talk_scene_flags = yes
				random_in_list = {
					variable = talk_scene_flags
					set_global_variable = {
						name = talk_scene
						value = this
					}
				}
				clear_variable_list = talk_scene_flags
				root = {
					trigger_event = mcc_event_v2.1010
				}
			}
		}
	}
}

#--------------------------------------------------------------------------
#  .9  Declined.  The NPC will not pester the player for a while; the length
#      of the grudge is opinion-scaled (votc_ce_initiative_decline_grudge).
#--------------------------------------------------------------------------
votc_ce_initiative.9 = {
	type = character_event
	hidden = yes
	orphan = yes

	immediate = {
		# VOTC-194 §34: the decline now happens inside the merged setup dialog
		# (mcc_event_v2.1010), so the -5 opinion the old .2 prompt applied is
		# applied here - the single place a declined invitation is handled.
		# scope:recipient is the NPC who reached out (saved by .1010).
		if = {
			limit = { exists = scope:recipient }
			scope:recipient = {
				add_opinion = {
					target = root
					modifier = conversation_opinion
					opinion = -5
				}
			}
		}
		remove_global_variable = votc_ce_initiative_player
		votc_ce_initiative_decline_grudge = yes
	}
}

#--------------------------------------------------------------------------
#  .152  A letter arrived through the NPC LETTER channel.
#        The script in run/votc_ce_npcltr.txt - written by the patched desktop app
#        and executed by gui/custom_gui/votc_ce_npc_letter_runner.gui (plus
#        the extended letters_runner) - fires this.  Hidden: no clipboard
#        write, no letter parchment, nothing that belongs to the player's own
#        letters.
#--------------------------------------------------------------------------
votc_ce_initiative.152 = {
	type = character_event
	hidden = yes
	orphan = yes

	immediate = {
		# VOTC-194 §30.1.2: the npcltr run file already created the artifact and
		# set votc_latest_letter, so hand the letter to the stock parchment
		# (message_event.362) and arm the reply state - closing the parchment then
		# opens the compose dialog in reply mode.  message_second_scope_npcltr is
		# the sender the app set on the request; it is consumed here (the .153
		# after-cleanup that used to release it is gone).
		if = {
			limit = { exists = global_var:message_second_scope_npcltr }
			set_global_variable = { name = votc_letter_reply_sender value = global_var:message_second_scope_npcltr }
		}
		remove_global_variable = message_second_scope_npcltr
		set_global_variable = { name = votc_letter_replyable value = yes }
		trigger_event = message_event.362
	}
}

#--------------------------------------------------------------------------
#  .180  Letter tools page (VOTC-194 §19/§20).  Clear pending incoming
#       letters, open the detailed per-topic settings window, send a test
#       letter, and release stuck letter threads.  The master toggle and the
#       unified frequency presets live in
#       gui/custom_gui/votc_incoming_letters_window.gui (§19); the test-letter
#       option was restored here per §20 (it also remains in the window).
#--------------------------------------------------------------------------
votc_ce_initiative.180 = {
	type = character_event
	theme = diplomacy_family_focus
	title = votc_ce_initiative.180.title
	desc = votc_ce_initiative.180.desc
	window = character_event

	immediate = {
		# VOTC-194 §18/§21: seed the letter frequencies as PLAYER VARIABLES
		# (every consumer reads var:votc_incoming_letters_*_frequency on the
		# player scope - the old global write here was a dead knob).  Each
		# frequency is only set when absent, so an existing player choice is
		# never overwritten (mirrors the only-when-unset shape of
		# votc_incoming_letters_apply_default_settings_effect).  An explicit
		# opt-out marker is honored: never re-enable behind the player's back.
		# votc_incoming_letters_enabled is NOT seeded here - the enable path
		# owns it.
		if = {
			limit = { NOT = { has_variable = votc_incoming_letters_opt_out } }
			if = {
				limit = { NOT = { has_variable = votc_incoming_letters_frequency } }
				set_variable = {
					name = votc_incoming_letters_frequency
					value = 2
				}
			}
			if = {
				limit = { NOT = { has_variable = votc_incoming_letters_event_frequency } }
				set_variable = {
					name = votc_incoming_letters_event_frequency
					value = 2
				}
			}
			if = {
				limit = { NOT = { has_variable = votc_incoming_letters_family_frequency } }
				set_variable = {
					name = votc_incoming_letters_family_frequency
					value = 2
				}
			}
			if = {
				limit = { NOT = { has_variable = votc_incoming_letters_poetry_frequency } }
				set_variable = {
					name = votc_incoming_letters_poetry_frequency
					value = 2
				}
			}
		}
	}

	# VOTC-194 14.3.2: clear every pending incoming-letter slot.
	option = {
		name = votc_ce_initiative_opt_incoming_clear
		trigger_event = votc_ce_initiative.186
	}
	# VOTC-194 14.3.3: detailed per-topic settings.  Reuses the existing opener
	# widget + votc_incoming_letters_window.gui - no new UI.
	option = {
		name = votc_ce_initiative_opt_incoming_details
		trigger_event = votc_letters_settings_bridge_event.9001
	}
	# VOTC-194 §20: quick-access test letter restored on this page; the button
	# also remains inside votc_incoming_letters_window.gui.
	option = {
		name = votc_ce_initiative_opt_letter_test
		trigger_event = votc_ce_initiative.174
	}
	option = {
		name = votc_ce_initiative_opt_release_letters
		custom_tooltip = votc_ce_initiative_opt_release_letters_tooltip
		trigger_event = votc_ce_initiative.200
	}
	option = {
		name = votc_ce_initiative_opt_back
		trigger_event = votc_ce_initiative.100
	}
	option = {
		name = votc_ce_initiative_opt_close
	}
}
