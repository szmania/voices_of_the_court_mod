message_event.360 = {
	type = character_event
	title = votc_message_title
	desc = votc_message_desc
	theme = diplomacy
	left_portrait = {
		character = scope:recipient
	}
	# VOTC-194: right-side portrait for the compose dialog.  In reply mode the
	# sender (votc_letter_reply_sender, set by votc_letter_parchment_closed_sgui
	# BEFORE this event is triggered) is shown on the right; in fresh-letter mode
	# the immediate below defaults the global to scope:recipient.  The immediate
	# runs before the window renders, so the global is always set by the time the
	# portrait block evaluates.

		character = global_var:votc_message_right_portrait
	}
	#orphan = yes

	widget = {
		gui = "event_window_widget_message_write"
		container = "custom_widgets_container"
		controller = {
			type = text
			data = {
				key = votc_message_text
			}
		}
		setup_scope = {
			root = { save_scope_as = text_target }
		}
	}

		set_global_variable = {
			name = message_first_scope
			value = scope:actor
		}
		set_global_variable = {
			name = message_second_scope
			value = scope:recipient
		}
		# VOTC-194: right-portrait target.  Default to the compose recipient;
		# in reply mode override with the letter's actual sender so the portrait
		# matches the "Reply to Letter from X" title.
		set_global_variable = {
			name = votc_message_right_portrait
			value = scope:recipient
		}
		if = {
			limit = { has_global_variable = votc_letter_reply_sender }
			set_global_variable = {
				name = votc_message_right_portrait
				value = global_var:votc_letter_reply_sender
			}
		}
		# VOTC-194: title hardening - if the compose was opened in reply mode but
		# the reply-sender global is somehow unset (e.g. an older save where the
		# sgui fallback path did not run), re-derive it from message_second_scope,
		# which in reply mode IS the letter's sender (the sgui saves the sender as
		# scope:recipient before triggering this event).  The title
		# (votc_letter_reply_title) reads this global at window-render time, after
		# this immediate, so the value is current when it evaluates.
		if = {
			limit = {
				has_global_variable = votc_letter_replying
				NOT = { has_global_variable = votc_letter_reply_sender }
				exists = global_var:message_second_scope
			}
			set_global_variable = {
				name = votc_letter_reply_sender
				value = global_var:message_second_scope
			}
		}
		# Testing distance calculation for days
		set_global_variable = {
			name = votc_letter_distance
			value = votc_letter_delay
		}

		if = {
			limit = {
				not = { exists = global_var:votc_letter_1 }
			}
			set_global_variable = votc_letter_1
			set_global_variable = {
				name = votc_letter_num
				value = flag:letter_1
			}
			set_global_variable = {
				name = message_second_scope_letter_1
				value = scope:recipient
			}
		}
		else_if = {
			limit = {
				not = { exists = global_var:votc_letter_2 }
			}
			# When variable letter2 does not exist, set letter2 as the current thread
			set_global_variable = votc_letter_2
			set_global_variable = {
				name = votc_letter_num
				value = flag:letter_2
			}
			set_global_variable = {
				name = message_second_scope_letter_2
				value = scope:recipient
			}
		}
		else_if = {
			limit = {
				not = { exists = global_var:votc_letter_3 }
			}
			# When variable letter3 does not exist, set letter3 as the current thread
			set_global_variable = votc_letter_3
			set_global_variable = {
				name = votc_letter_num
				value = flag:letter_3
			}
			set_global_variable = {
				name = message_second_scope_letter_3
				value = scope:recipient
			}
		}
		else_if = {
			limit = {
				not = { exists = global_var:votc_letter_4 }
			}
			# When variable letter4 does not exist, set letter4 as the current thread
			set_global_variable = votc_letter_4
			set_global_variable = {
				name = votc_letter_num
				value = flag:letter_4
			}
			set_global_variable = {
				name = message_second_scope_letter_4
				value = scope:recipient
			}
		}
		else_if = {
			limit = {
				not = { exists = global_var:votc_letter_5 }
			}
			# When variable letter5 does not exist, set letter5 as the current thread
			set_global_variable = votc_letter_5
			set_global_variable = {
				name = votc_letter_num
				value = flag:letter_5
			}
			set_global_variable = {
				name = message_second_scope_letter_5
				value = scope:recipient
			}
		}
		else_if = {
			limit = {
				not = { exists = global_var:votc_letter_6 }
			}
			# When variable letter6 does not exist, set letter6 as the current thread
			set_global_variable = votc_letter_6
			set_global_variable = {
				name = votc_letter_num
				value = flag:letter_6
			}
			set_global_variable = {
				name = message_second_scope_letter_6
				value = scope:recipient
			}
			#debug_log = "VOTC:LETTER set to thread votc_letter_6"
		}
		else_if = {
			limit = {
				not = { exists = global_var:votc_letter_7 }
			}
			# When variable letter7 does not exist, set letter7 as the current thread
			set_global_variable = votc_letter_7
			set_global_variable = {
				name = votc_letter_num
				value = flag:letter_7
			}
			set_global_variable = {
				name = message_second_scope_letter_7
				value = scope:recipient
			}
			#debug_log = "VOTC:LETTER set to thread votc_letter_7"
		}
		else_if = {
			limit = {
				not = { exists = global_var:votc_letter_8 }
			}
			# When variable letter8 does not exist, set letter8 as the current thread
			set_global_variable = votc_letter_8
			set_global_variable = {
				name = votc_letter_num
				value = flag:letter_8
			}
			set_global_variable = {

				value = scope:recipient

		}
		else_if = {
			limit = {
				not = { exists = global_var:votc_letter_9 }
			}
			# When variable letter9 does not exist, set letter9 as the current thread
			set_global_variable = votc_letter_9
			set_global_variable = {

		remove_global_variable = mcc_character_1

		remove_variable ?= votc_incoming_reply_slot
		remove_variable ?= votc_incoming_reply_type
		remove_variable ?= votc_incoming_reply_delivery
		# VOTC-194 §30.3.3: a reply-mode send consumes the read artifact and ends
		# the reply turn; the chain continues when the NPC's next letter arrives as
		# a new artifact.  Fresh letters are untouched (marker unset).
		if = {
			limit = { has_global_variable = votc_letter_replying }
			remove_global_variable = votc_letter_replying
			# VOTC-194: clear the reply sender with the marker - a stale sender
			# must never leak into the next letter's "Reply to Letter from X" title.
			remove_global_variable = votc_letter_reply_sender
			votc_ce_initiative_letter_destroy = yes
		}

		#remove_variable = votc_message_text
	}

		# VOTC-194: dedicated Close key for the compose dialog.  The old shared
		# mcc_event.1002.exit key ("Leave") is left untouched for its other callers.
		name = votc_ce_initiative_letter_compose_close
		# NOTE: no `enabled` field exists on CK3 event options (ck3-tiger rejects it
		# as an unknown token).  An option with no `trigger` is always clickable,
		# so this Close button is already dismissable before any text is entered.
		# VOTC-194 §30.3.3: Close in reply mode = end the chain - consume the read
		# artifact and clear the reply marker (same cleanup as Send).
		if = {
			limit = { has_global_variable = votc_letter_replying }
			remove_global_variable = votc_letter_replying
			# VOTC-194: clear the reply sender with the marker - a stale sender
			# must never leak into the next letter's "Reply to Letter from X" title.
			remove_global_variable = votc_letter_reply_sender
			votc_ce_initiative_letter_destroy = yes
		}
	}
}

# Opens letter widget
message_event.362 = {
	type = character_event
	title = votc_placeholder
	theme = realm
	desc = message_event.362.desc
	window = fullscreen_event
	widget = {
        gui = "event_window_widget_message_open"
        container = "custom_widgets_container"
    }

	# VOTC-195: The Whisper Ledger fallback channel.  While the desktop app has
	# no whisper branch yet, a whisper reply arrives through the ordinary letter
	# channel (run/letters.txt) and is hijacked into the whisper pipeline here.
	# The hook is fully guarded: it only fires when a whisper sequence is armed
	# (votc_whisper_pending), the app has already deleted the request marker
	# (votc_whisper gone), the report is not finished yet (votc_whisper_done
	# unset), and both the reply artifact (votc_latest_letter) and the player
	# scope (votc_whisper_player) exist.  An ordinary letter never satisfies all
	# five conditions at once, so the letter flow is untouched.
	immediate = {
		votc_whisper_deliver = yes
		# VOTC-194 §30.1.3 + §34: arm the reply state for legacy letter replies.
		# When the artifact exists and this is NOT a whisper-hijacked reply,
		# closing the parchment opens the compose dialog in reply mode.  The
		# reply sender is ALWAYS re-derived on landing: the old
		# not-already-set guard let a sender from a PREVIOUS letter cycle
		# survive into this one, so closing a reply from X opened
		# "Reply to Letter from Y".  The npcltr path (.152) mirrors its
		# sender into message_second_scope before triggering this event, so
		# the derivation below is correct for every channel that lands here.
		if = {
			limit = {
				exists = global_var:votc_latest_letter
				NOT = { has_global_variable = votc_whisper_pending }
			}
			set_global_variable = { name = votc_letter_replyable value = yes }
			remove_global_variable ?= votc_letter_reply_sender
			# VOTC-194 §39: prefer the ARTIFACT CREATOR (scope:artifact.creator,
			# the stock artifact->character event target) - the app's
			# create_artifact sets creator = the letter's NPC sender, so this is
			# intrinsic to THIS letter and immune to a stale message_second_scope
			# left by an earlier compose.  message_second_scope stays as the
			# fallback for artifacts without a creator (older saves).
			if = {
				limit = {
					exists = global_var:votc_latest_letter.creator
				}
				global_var:votc_latest_letter.creator = {
					save_scope_as = votc_letter_reply_target
				}
				set_global_variable = { name = votc_letter_reply_sender value = scope:votc_letter_reply_target }
			}
			else_if = {
				limit = { exists = global_var:message_second_scope }
				set_global_variable = { name = votc_letter_reply_sender value = global_var:message_second_scope }
			}
		}
	}
}
