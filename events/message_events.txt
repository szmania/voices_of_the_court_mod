# Opens letter widget
message_event.362 = {
	type = character_event
	title = votc_placeholder
	theme = realm
	desc = message_event.362.desc
	window = fullscreen_event
	widget = {
        gui = "event_window_widget_message_open"
        container = "custom_widgets_container"
    }

	# VOTC-195: The Whisper Ledger fallback channel.  While the desktop app has
	# no whisper branch yet, a whisper reply arrives through the ordinary letter
	# channel (run/letters.txt) and is hijacked into the whisper pipeline here.
	# The hook is fully guarded: it only fires when a whisper sequence is armed
	# (votc_whisper_pending), the app has already deleted the request marker
	# (votc_whisper gone), the report is not finished yet (votc_whisper_done
	# unset), and both the reply artifact (votc_latest_letter) and the player
	# scope (votc_whisper_player) exist.  An ordinary letter never satisfies all
	# five conditions at once, so the letter flow is untouched.
	immediate = {
		votc_whisper_deliver = yes
		# VOTC-194 §30.1.3: arm the reply state for legacy letter replies.  When
		# the artifact exists and this is NOT a whisper-hijacked reply, closing the
		# parchment opens the compose dialog in reply mode.  votc_letter_reply_sender
		# is normally set by the npcltr (.152) or incoming (.010N) path already;
		# message_second_scope is the safe legacy fallback (set by the player's own
		# message_event.360 send).
		if = {
			limit = {
				exists = global_var:votc_latest_letter
				NOT = { has_global_variable = votc_whisper_pending }
			}
			set_global_variable = { name = votc_letter_replyable value = yes }
			if = {
				limit = {
					NOT = { has_global_variable = votc_letter_reply_sender }
					exists = global_var:message_second_scope
				}
				set_global_variable = { name = votc_letter_reply_sender value = global_var:message_second_scope }
			}
		}
	}
}
