# VOTC-194: NPC Initiative - scripted effects (CE-native rewrite).
# root = the player character; scope:recipient = the NPC (VOTC convention).

# VOTC-194 13.1.2: shared quarterly letter budget.  Counted across BOTH
# letter systems (NPC Initiative npcltr + votc_incoming_letter slots),
# checked against votc_letter_quarter_cap by both pickers.  The counter is
# a timed global (91 days) so a stray count can never outlive its quarter;
# re-seeding keeps the rolling quarter alive.
votc_ce_letter_budget_increment = {
	# VOTC-194: silently auto-initialize the rolling quarter counter if it
	# expired or was never created, so the increment below can never read an
	# unset global ("Failed to fetch variable" / "Invalid left side" errors).
	if = {
		limit = { NOT = { has_global_variable = votc_letters_this_quarter } }
		set_global_variable = {
			name = votc_letters_this_quarter
			value = 0
			days = 91
		}
	}
	change_global_variable = {
		name = votc_letters_this_quarter
		add = 1
	}
}

votc_ce_initiative_bootstrap_char = {
	if = {
		limit = {
			is_ai = no
			is_alive = yes
		}
		if = {
			limit = { NOT = { has_global_variable = votc_ce_initiative_freq } }
			set_global_variable = { name = votc_ce_initiative_freq value = 182 }
		}
		if = {
			limit = { NOT = { has_global_variable = votc_ce_initiative_chance } }
			set_global_variable = { name = votc_ce_initiative_chance value = 15 }
		}
		if = {
			limit = { NOT = { has_global_variable = votc_ce_initiative_letter_freq } }
			set_global_variable = { name = votc_ce_initiative_letter_freq value = 182 }
		}
		# VOTC-194 §17: ongoing-save default-on for incoming letters.  New
		# campaigns are covered by the game-start hook (votc_game_start_init_relay);
		# this mirrors it for saves that predate the feature.  The opt-out marker
		# keeps an explicit "off" from being reverted.
		if = {
			limit = {
				NOT = { has_variable = votc_incoming_letters_enabled }
				NOT = { has_variable = votc_incoming_letters_opt_out }
			}
			votc_incoming_letters_apply_default_settings_effect = yes
		}
		# VOTC-194 13.1.2: shared quarterly letter budget across the NPC
		# Initiative and incoming-letter systems.  Default allowance: 2 letters
		# per quarter from both systems combined.
		if = {
			limit = { NOT = { has_global_variable = votc_letter_quarter_cap } }
			set_global_variable = { name = votc_letter_quarter_cap value = 2 }
		}
		# VOTC-194: keep the rolling quarter counter alive on ongoing saves -
		# the quarterly watchdog re-seeds it whenever the 91-day expiry lapses
		# (a 365-day talk cadence would otherwise leave it unset for months).
		if = {
			limit = { NOT = { has_global_variable = votc_letters_this_quarter } }
			set_global_variable = {
				name = votc_letters_this_quarter
				value = 0
				days = 91
			}
		}
		if = {
			limit = { NOT = { exists = global_var:votc_ce_initiative_heartbeat } }
			trigger_event = { id = votc_ce_initiative.1 days = 15 }
		}
		votc_ce_initiative_purge_lists = yes
		votc_ce_initiative_release_stale = yes
		if = {
			limit = { NOT = { exists = global_var:votc_ce_initiative_letter_heartbeat } }
			trigger_event = { id = votc_ce_initiative.5 days = 10 }
		}
	}
}

votc_ce_initiative_purge_lists = {
	if = {
		limit = { has_variable_list = votc_talked_characters }
		if = {
			limit = { has_variable_list = votc_ce_initiative_tmp }
			clear_variable_list = votc_ce_initiative_tmp
		}
		every_in_list = {
			variable = votc_talked_characters
			limit = { is_alive = yes }
			save_temporary_scope_as = votc_ce_purge_keep
			root = {
				add_to_variable_list = { name = votc_ce_initiative_tmp target = scope:votc_ce_purge_keep }
			}
		}
		clear_variable_list = votc_talked_characters
		if = {
			limit = {
				has_variable_list = votc_ce_initiative_tmp
				variable_list_size = { name = votc_ce_initiative_tmp value >= 1 }
			}
			every_in_list = {
				variable = votc_ce_initiative_tmp
				save_temporary_scope_as = votc_ce_purge_keep
				root = {
					add_to_variable_list = { name = votc_talked_characters target = scope:votc_ce_purge_keep }
				}
			}
		}
		clear_variable_list = votc_ce_initiative_tmp
	}
	if = {
		limit = { has_variable_list = votc_written_characters }
		if = {
			limit = { has_variable_list = votc_ce_initiative_tmp }
			clear_variable_list = votc_ce_initiative_tmp
		}
		every_in_list = {
			variable = votc_written_characters
			limit = { is_alive = yes }
			save_temporary_scope_as = votc_ce_purge_keep
			root = {
				add_to_variable_list = { name = votc_ce_initiative_tmp target = scope:votc_ce_purge_keep }
			}
		}
		clear_variable_list = votc_written_characters
		if = {
			limit = {
				has_variable_list = votc_ce_initiative_tmp
				variable_list_size = { name = votc_ce_initiative_tmp value >= 1 }
			}
			every_in_list = {
				variable = votc_ce_initiative_tmp
				save_temporary_scope_as = votc_ce_purge_keep
				root = {
					add_to_variable_list = { name = votc_written_characters target = scope:votc_ce_purge_keep }
				}
			}
		}
		clear_variable_list = votc_ce_initiative_tmp
	}
}

votc_ce_initiative_release_stale = {
	if = {
		limit = {
			has_global_variable = votc_ce_initiative_player
			OR = {
				NOT = { exists = global_var:votc_ce_initiative_char }
				global_var:votc_ce_initiative_char = { NOT = { is_alive = yes } }
			}
		}
		debug_log = "VOTC_CE_INIT: the invited character is dead or gone - releasing the invitation"
		remove_global_variable = votc_ce_initiative_player
		remove_global_variable = votc_ce_initiative_char
	}
	if = {
		limit = {
			has_global_variable = votc_ce_initiative_player
			exists = global_var:votc_ce_initiative_char
			global_var:votc_ce_initiative_char = {
				is_alive = yes
				is_travelling = no
				NOT = { is_at_same_location = root }
			}
		}
		# VOTC-194 §57.3.2: self-heal - the invited NPC is alive, NOT travelling
		# and NOT at the player's location, so no arrival event (.4) can ever
		# fire (the travel plan was silently cancelled).  Release the wedged
		# invitation instead of letting it block the feature until the 730-day
		# hard expiry on votc_ce_initiative_char lapses.
		debug_log = "VOTC_CE_INIT: invited character is stranded (not travelling, not co-located) - releasing the wedged invitation"
		remove_global_variable = votc_ce_initiative_player
		remove_global_variable = votc_ce_initiative_char
	}
	if = {
		limit = {
			has_global_variable = votc_ce_initiative_letter_pending
			OR = {
				NOT = { exists = global_var:votc_ce_initiative_letter_char }
				global_var:votc_ce_initiative_letter_char = { NOT = { is_alive = yes } }
			}
		}
		debug_log = "VOTC_CE_LETTER: the writer is dead or gone - releasing the in-flight guard"
		votc_ce_initiative_letter_cleanup = yes
	}
	if = {
		limit = {
			has_global_variable = votc_ce_initiative_letter_pending
			NOT = { has_global_variable = votc_npcltr }
			exists = global_var:votc_ce_initiative_letter_char
			global_var:votc_ce_initiative_letter_char = { is_alive = yes }
		}
		debug_log = "VOTC_CE_LETTER: reply landed through the normal letter channel (unpatched app) - releasing the in-flight guard"
		votc_ce_initiative_letter_cleanup = yes
	}
}

votc_ce_initiative_try = {
	votc_ce_initiative_purge_lists = yes
	votc_ce_initiative_release_stale = yes
	# VOTC-194 §57.3.4: seed the pool from pinned characters BEFORE the
	# has_variable_list gate, so a new player with pins but no completed
	# VOTC talks is not locked out (chicken-and-egg fix).
	votc_ce_initiative_merge_watch = yes
	# VOTC-194 §57.3.1: pre-gate breadcrumb - records the gate state on EVERY
	# entry so a silent no-fire is diagnosable from debug.log.
	if = {
		limit = {
			is_ai = no
			is_alive = yes
			NOT = { is_imprisoned = yes }
			NOT = { is_travelling = yes }
			has_variable_list = votc_talked_characters
			NOT = { exists = global_var:votc_ce_initiative_player }
		}
		debug_log = "VOTC_CE_INIT: try gates passed - rolling chance"
		if = {
			limit = { global_var:votc_ce_initiative_chance = 5 }
			random = { chance = 5 votc_ce_initiative_launch = yes }
		}
		else_if = {
			limit = { global_var:votc_ce_initiative_chance = 15 }
			random = { chance = 15 votc_ce_initiative_launch = yes }
		}
		else_if = {
			limit = { global_var:votc_ce_initiative_chance = 30 }
			random = { chance = 30 votc_ce_initiative_launch = yes }
		}
		else = {
			random = { chance = 15 votc_ce_initiative_launch = yes }
	}
	}
	# VOTC-194 §57.3.1: name the exact blocked gate so a silent no-fire is
	# diagnosable from debug.log instead of guesswork.
	else = {
		debug_log = "VOTC_CE_INIT: try gates blocked - mcc_event_v2.1010 not fired"
		if = {
			limit = { NOT = { has_variable_list = votc_talked_characters } }
			debug_log = "VOTC_CE_INIT: gate fail - votc_talked_characters pool empty"
		}
		if = {
			limit = { exists = global_var:votc_ce_initiative_player }
			debug_log = "VOTC_CE_INIT: gate fail - an invitation is already pending"
		}
		if = {
			limit = { has_global_variable = votc_ce_initiative_off }
			debug_log = "VOTC_CE_INIT: gate fail - feature disabled (votc_ce_initiative_off)"
		}
		if = {
			limit = { is_imprisoned = yes }
			debug_log = "VOTC_CE_INIT: gate fail - the player is imprisoned"
		}
		if = {
			limit = { is_travelling = yes }
			debug_log = "VOTC_CE_INIT: gate fail - the player is travelling"
		}
	}
}

votc_ce_initiative_merge_watch = {
	root = {
		every_pinned_character = {
			if = {
				limit = {
					is_alive = yes
					is_adult = yes
					is_ai = yes
					NOT = { this = root }
				}
				save_temporary_scope_as = votc_ce_initiative_pinned
				root = {
					if = {
						limit = { NOT = { has_variable_list = votc_talked_characters } }
						add_to_variable_list = { name = votc_talked_characters target = scope:votc_ce_initiative_pinned }
					}
					else_if = {
						limit = {
							NOT = {
								is_target_in_variable_list = { name = votc_talked_characters target = scope:votc_ce_initiative_pinned }
							}
						}
						add_to_variable_list = { name = votc_talked_characters target = scope:votc_ce_initiative_pinned }
					}
				}
			}
		}
	}
}

votc_ce_initiative_launch = {
	if = {
		limit = { NOT = { exists = global_var:votc_ce_initiative_player } }
		save_scope_as = actor
		# VOTC-194 §57.3.1: breadcrumb before the same-location pick so a
		# silent no-fire is diagnosable from debug.log.
		debug_log = "VOTC_CE_INIT: launch: rolling same-location pick over votc_talked_characters"
		ordered_in_list = {
			variable = votc_talked_characters
			limit = {
				votc_ce_initiative_is_valid_candidate = yes
				is_at_same_location = root
			}
			order_by = votc_ce_initiative_weight
			max = 1
			save_scope_as = votc_ce_initiative_target
		}
		if = {
			limit = { NOT = { exists = scope:votc_ce_initiative_target } }
			# VOTC-194 §57.3.1: breadcrumb before the fallback any-location pick.
			debug_log = "VOTC_CE_INIT: launch: no same-location candidate - rolling any-location pick"
			ordered_in_list = {
				variable = votc_talked_characters
				limit = { votc_ce_initiative_is_valid_candidate = yes }
				order_by = votc_ce_initiative_weight
				max = 1
				save_scope_as = votc_ce_initiative_target
			}
		}
		if = {
			limit = { exists = scope:votc_ce_initiative_target }
			# VOTC-194 §57.3.2: hard expiry so a wedged invitation (e.g. a
			# silently cancelled travel plan) cannot block the feature forever.
			set_global_variable = { name = votc_ce_initiative_char value = scope:votc_ce_initiative_target days = 730 }
			set_global_variable = { name = votc_ce_initiative_player value = root days = 90 }
			scope:votc_ce_initiative_target = {
				if = {
					limit = { opinion = { target = root value >= 25 } }
					set_global_variable = { name = votc_ce_initiative_mood value = flag:warm }
				}
				else_if = {
					limit = { opinion = { target = root value <= -25 } }
					set_global_variable = { name = votc_ce_initiative_mood value = flag:cold }
				}
				else = {
					set_global_variable = { name = votc_ce_initiative_mood value = flag:neutral }
				}
			}
			debug_log = "VOTC_CE_INIT: picked [scope:votc_ce_initiative_target.GetID] to request a talk"
			trigger_event = votc_ce_initiative.3
		}
		else = {
			debug_log = "VOTC_CE_INIT: launch: no valid candidate in pool - mcc_event_v2.1010 not fired"
			# VOTC-194 §57.3.1: the stuck-talk_scene disqualifier (see
			# votc_ce_initiative_is_valid_candidate) silently removes every
			# candidate - name it explicitly so the wedge is visible.
			if = {
				limit = { has_global_variable = talk_scene }
				debug_log = "VOTC_CE_INIT: gate fail - talk_scene is set (stuck conversation?) disqualifying all candidates"
			}
		}
	}
}

votc_ce_initiative_record = {
	if = {
		limit = { exists = scope:recipient }
		scope:recipient = {
			debug_log = "VOTC_CE_INIT: recorded a talk with [scope:recipient.GetID]"
			if = {
				limit = { has_character_flag = votc_ce_initiative_declined }
				remove_character_flag = votc_ce_initiative_declined
			}
		}
		if = {
			limit = { root = { is_ai = no } }
			root = {
				if = {
					limit = { has_variable_list = votc_talked_characters }
					if = {
						limit = {
							NOT = {
								is_target_in_variable_list = { name = votc_talked_characters target = scope:recipient }
							}
						}
						add_to_variable_list = { name = votc_talked_characters target = scope:recipient }
					}
				}
				else = {
					add_to_variable_list = { name = votc_talked_characters target = scope:recipient }
				}
			}
		}
	}
	else = {
		debug_log = "VOTC_CE_INIT: no recipient scope - talk not recorded"
	}
}

# VOTC-194 13.2.2: an incoming letter's author joins the NPC-Initiative
# pools so relationship weighting learns from both channels.  root = the
# player, scope:recipient = the incoming letter's sender.  Talks and written
# letters are both pools the pickers read, so the sender becomes reachable
# by both; duplicates are no-ops.
votc_ce_initiative_record_letter_sender = {
	if = {
		limit = { exists = scope:recipient }
		scope:recipient = {
			if = {
				limit = { has_character_flag = votc_ce_initiative_declined }
				remove_character_flag = votc_ce_initiative_declined
			}
		}
		scope:actor = {
			if = {
				limit = { has_variable_list = votc_talked_characters }
				if = {
					limit = {
						NOT = {
							is_target_in_variable_list = { name = votc_talked_characters target = scope:recipient }
						}
					}
					add_to_variable_list = { name = votc_talked_characters target = scope:recipient }
				}
			}
			else = {
				add_to_variable_list = { name = votc_talked_characters target = scope:recipient }
			}
		}
		scope:actor = {
			if = {
				limit = { has_variable_list = votc_written_characters }
				if = {
					limit = {
						NOT = {
							is_target_in_variable_list = { name = votc_written_characters target = scope:recipient }
						}
					}
					add_to_variable_list = { name = votc_written_characters target = scope:recipient }
				}
			}
			else = {
				add_to_variable_list = { name = votc_written_characters target = scope:recipient }
			}
		}
	}
}

votc_ce_initiative_letter_pick = {
	votc_ce_initiative_release_stale = yes
	votc_ce_initiative_build_letter_pool = yes
	# VOTC-194: seed the rolling quarter counter if it expired or was never
	# created (fresh campaign; the 91-day expiry can lapse between heartbeats)
	# so the shared-budget comparison below never reads an unset global.
	if = {
		limit = { NOT = { has_global_variable = votc_letters_this_quarter } }
		set_global_variable = {
			name = votc_letters_this_quarter
			value = 0
			days = 91
		}
	}
	if = {
		limit = {
			has_variable_list = votc_ce_initiative_letter_pool
			variable_list_size = { name = votc_ce_initiative_letter_pool value >= 1 }
			# VOTC-194 13.1.1: an in-flight incoming-letter slot suppresses a
			# same-quarter NPC-Initiative letter (mirror of the votc_npcltr gate
			# on votc_incoming_letter_event.0001).
			NOT = { has_global_variable = votc_incoming_letter_1 }
			NOT = { has_global_variable = votc_incoming_letter_2 }
			NOT = { has_global_variable = votc_incoming_letter_3 }
			NOT = { has_global_variable = votc_incoming_letter_4 }
			NOT = { has_global_variable = votc_incoming_letter_5 }
			NOT = { has_global_variable = votc_incoming_letter_6 }
			NOT = { has_global_variable = votc_incoming_letter_7 }
			NOT = { has_global_variable = votc_incoming_letter_8 }
			# VOTC-194 13.1.2: shared quarterly letter budget across both
			# systems.  The OR-guard keeps an unset cap (pre-bootstrap) from
			# freezing the picker (unset globals compare as 0 >= 0 -> never fire).
			OR = {
				NOT = { exists = global_var:votc_letter_quarter_cap }
				NOT = { global_var:votc_letters_this_quarter >= global_var:votc_letter_quarter_cap }
			}
		}
		save_scope_as = actor
		if = {
			limit = {
				exists = global_var:votc_ce_initiative_last_writer
				variable_list_size = { name = votc_ce_initiative_letter_pool value >= 2 }
			}
			ordered_in_list = {
				variable = votc_ce_initiative_letter_pool
				limit = {
					votc_ce_initiative_is_letter_candidate = yes
					NOT = { this = global_var:votc_ce_initiative_last_writer }
				}
				order_by = votc_ce_initiative_weight
				max = 1
				save_scope_as = votc_ce_initiative_letter
			}
		}
		if = {
			limit = { NOT = { exists = scope:votc_ce_initiative_letter } }
			ordered_in_list = {
				variable = votc_ce_initiative_letter_pool
				limit = { votc_ce_initiative_is_letter_candidate = yes }
				order_by = votc_ce_initiative_weight
				max = 1
				save_scope_as = votc_ce_initiative_letter
			}
		}
		if = {
			limit = { exists = scope:votc_ce_initiative_letter }
			set_global_variable = { name = votc_ce_initiative_letter_char value = scope:votc_ce_initiative_letter }
			set_global_variable = { name = votc_ce_initiative_letter_player value = root days = 400 }
			set_global_variable = { name = votc_ce_initiative_letter_pending value = yes days = 400 }
			set_global_variable = { name = votc_ce_initiative_last_writer value = scope:votc_ce_initiative_letter days = 400 }
			debug_log = "VOTC_CE_LETTER: picked [scope:votc_ce_initiative_letter.GetID] to write"
			root = { votc_ce_initiative_letter = yes }
		}
	}
}

		limit = { has_variable_list = votc_written_characters }

			limit = { is_alive = yes }
			save_temporary_scope_as = votc_ce_lp_target
			root = {

							is_target_in_variable_list = { name = votc_ce_initiative_letter_pool target = scope:votc_ce_lp_target }
						}
					}
					add_to_variable_list = { name = votc_ce_initiative_letter_pool target = scope:votc_ce_lp_target }
				}
			}
		}
	}
	root = {
		every_pinned_character = {
			if = {

					is_alive = yes
					is_adult = yes
					is_ai = yes
					NOT = { this = root }
				}
				save_temporary_scope_as = votc_ce_lp_target
				root = {
					if = {
						limit = {
							NOT = {
								is_target_in_variable_list = { name = votc_ce_initiative_letter_pool target = scope:votc_ce_lp_target }
							}
						}
						add_to_variable_list = { name = votc_ce_initiative_letter_pool target = scope:votc_ce_lp_target }
					}
				}
			}
		}
	}
}

votc_ce_initiative_letter = {
	if = {
		limit = { exists = global_var:votc_ce_initiative_letter_char }
		global_var:votc_ce_initiative_letter_char = {
			save_temporary_scope_as = votc_ce_initiative_sender
		}
		if = {
			limit = { exists = global_var:votc_ce_initiative_letter_player }
			global_var:votc_ce_initiative_letter_player = {
				save_temporary_scope_as = votc_ce_initiative_recipient
			}
		}
		else = {
			root = { save_temporary_scope_as = votc_ce_initiative_recipient }
		}
		scope:votc_ce_initiative_recipient = { save_scope_as = actor }
		scope:votc_ce_initiative_sender = {
			debug_log = "VOTC:IN/;/init/;/[GetPlayer.GetID]/;/[GetPlayer.GetTitledFirstNameNicknamedNoTooltipRegnal]/;/[THIS.Char.GetID]/;/[THIS.Char.GetTitledFirstNameNicknamedNoTooltipRegnal]/;/[GetCurrentDate.GetStringShort]/;/flag:talk_letter/;/[THIS.Char.GetCurrentLocation.GetName]/;/none/;/[GetCurrentDate.GetDateAsTotalDays]/;/none"
		}
		if = {
			limit = { exists = global_var:votc_ce_initiative_letter_player }
			global_var:votc_ce_initiative_letter_player = { log_character_v2 = yes }
		}
		else = {
			root = { log_character_v2 = yes }
		}
		scope:votc_ce_initiative_sender = {
			log_character_v2 = yes
			if = {
				limit = {
					OR = {
						has_relation_lover = scope:votc_ce_initiative_recipient
						has_relation_soulmate = scope:votc_ce_initiative_recipient
					}
				}
				debug_log = votc_ce_initiative_log_letter_lover
			}
			else_if = {
				limit = {
					OR = {
						has_relation_friend = scope:votc_ce_initiative_recipient
						has_relation_best_friend = scope:votc_ce_initiative_recipient
					}
				}
				debug_log = votc_ce_initiative_log_letter_friend
			}
			else_if = {
				limit = {
					OR = {
						has_relation_rival = scope:votc_ce_initiative_recipient
						has_relation_nemesis = scope:votc_ce_initiative_recipient
					}
				}
				debug_log = votc_ce_initiative_log_letter_rival
			}
			else_if = {
				limit = {
					OR = {
						is_parent_of = scope:votc_ce_initiative_recipient
						is_child_of = scope:votc_ce_initiative_recipient
						is_sibling_of = scope:votc_ce_initiative_recipient
					}
				}
				debug_log = votc_ce_initiative_log_letter_kin
			}
			else_if = {
				limit = {
					OR = {
						is_liege_or_above_of = scope:votc_ce_initiative_recipient
						is_vassal_of = scope:votc_ce_initiative_recipient
					}
				}
				debug_log = votc_ce_initiative_log_letter_liege
			}
			else_if = {
				limit = {
					OR = {
						has_relation_guardian = scope:votc_ce_initiative_recipient
						has_relation_ward = scope:votc_ce_initiative_recipient
					}
				}
				debug_log = votc_ce_initiative_log_letter_mentor
			}
			else = {
				debug_log = votc_ce_initiative_log_letter_plain
			}
		}
		set_global_variable = { name = message_second_scope_npcltr value = scope:votc_ce_initiative_sender }
		set_global_variable = { name = votc_npcltr value = yes days = 400 }
		# VOTC-194 13.1.2: this NPC letter spends one unit of the shared
		# quarterly letter budget.
		votc_ce_letter_budget_increment = yes
		trigger_event = { id = votc_ce_initiative.160 days = 1 }
	}
}

votc_ce_initiative_decline_grudge = {
	if = {
		limit = { exists = global_var:votc_ce_initiative_char }
		global_var:votc_ce_initiative_char = {
			if = {
				limit = { opinion = { target = root value <= 0 } }
				add_character_flag = { flag = votc_ce_initiative_declined days = 365 }
			}
			else = {
				add_character_flag = { flag = votc_ce_initiative_declined days = 182 }
			}
		}
	}
	remove_global_variable = votc_ce_initiative_player
}
