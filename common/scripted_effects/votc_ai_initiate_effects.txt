# An AI character at the player's location asks to speak with the player.
# The AI (scripted random selection) picks the initiator, 2-15 other
# participants at the location (75% of the time none), and the scene, then pops up a request event
# the player can accept or decline. Runs on the player's scope.
votc_ai_initiate_request_effect = {

	clear_variable_list = mcc_characters_selected

	# Build the AI candidate list at the player's location.
	location = {
		every_character_in_location = {
			limit = {
				votc_ai_can_initiate_conversation_trigger = yes
			}
			scope:actor = {
				add_to_variable_list = {
					name = votc_ai_initiate_candidates
					target = prev
				}
			}
		}
	}
	if = {
		limit = {
			has_variable_list = votc_ai_initiate_candidates
		}
		# The AI initiator.
		random_in_list = {
			variable = votc_ai_initiate_candidates
			save_scope_as = recipient
		}
		set_global_variable = {
			name = votc_ai_initiator
			value = scope:recipient
		}
		# The AI decides whether to bring company: 75% of the time it is just
		# the initiator and the player; 25% of the time it brings 2-15 other
		# eligible characters from the location (never the player). They land in
		# the same list the normal MCC flow uses.
		random = {
			chance = 25
			every_in_list = {
				variable = votc_ai_initiate_candidates
				limit = {
					NOT = { this = scope:recipient }
				}
				random = {
					chance = 50
					scope:actor = {
						if = {
							limit = {
								OR = {
									NOT = { has_variable_list = mcc_characters_selected }
									variable_list_size = {
										name = mcc_characters_selected
										value < 15
									}
								}
							}
							add_to_variable_list = {
								name = mcc_characters_selected
								target = prev
							}
						}
					}
				}
			}
			# Guarantee at least 2 extra participants when the 25% branch fires.
			if = {
				limit = {
					OR = {
						NOT = { has_variable_list = mcc_characters_selected }
						variable_list_size = {
							name = mcc_characters_selected
							value < 2
						}
					}
				}
				ordered_in_list = {
					variable = votc_ai_initiate_candidates
					limit = {
						NOT = { this = scope:recipient }
					}
					order_by = age
					min = 0
					max = 2
					scope:actor = {
						add_to_variable_list = {
							name = mcc_characters_selected
							target = prev
						}
					}
				}
			}
		}
		# The AI picks the scene from the same valid pool the player would see.
		votc_build_talk_scene_flags = yes
		random_in_list = {
			variable = talk_scene_flags
			set_global_variable = {
				name = talk_scene
				value = this
			}
		}
		clear_variable_list = talk_scene_flags
		# Cooldowns so neither side is asked again immediately.
		add_character_flag = { flag = votc_ai_initiate_cooldown days = 10 }
		scope:recipient = {
			add_character_flag = { flag = votc_ai_initiate_cooldown days = 10 }
		}
		clear_variable_list = votc_ai_initiate_candidates
		# Popup to the player: accept or decline the conversation.
		trigger_event = mcc_event_v2.1010
	}
	else = {

		clear_variable_list = votc_ai_initiate_candidates

	}

}



votc_ai_initiate_conversation_effect = {

	# Save the player so nested list access works from candidate scopes.

	save_scope_as = actor

	# 0-2 requests per month, randomized.

	random_list = {

		60 = {}

		30 = { votc_ai_initiate_request_effect = yes }

		10 = {

			votc_ai_initiate_request_effect = yes

			votc_ai_initiate_request_effect = yes

		}

	}

	# Re-arm the monthly tick so the loop sustains itself.

	if = {

		limit = { NOT = { has_character_flag = votc_ai_tick_armed } }

		add_character_flag = { flag = votc_ai_tick_armed days = 35 }

		trigger_event = { id = mcc_event_v2.1012 days = 30 }

	}

}
