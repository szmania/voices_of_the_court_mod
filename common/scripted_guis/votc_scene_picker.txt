select_scene = {
	scope = character
	effect = {
		set_global_variable = {
			name = talk_scene
			value = scope:scene
		}
		# Trigger the main conversation event now that the scene is selected
		if = {
			limit = { has_character_flag = is_mcc_convo }
			trigger_event = {
				id = mcc_event.9001
				actor = var:talk_actor
				recipient = var:talk_recipient
				secondary_recipient = var:talk_secondary_recipient
			}
			remove_character_flag = is_mcc_convo
		}
		else_if = {
			limit = { has_character_flag = is_meet_convo }
			trigger_event = {
				id = ckms_meet_event.9001
				actor = var:talk_actor
				recipient = var:talk_recipient
				secondary_recipient = var:talk_secondary_recipient
			}
			remove_character_flag = is_meet_convo
		}
		else = {
			trigger_event = {
				id = talk_event.9001
				actor = var:talk_actor
				recipient = var:talk_recipient
			}
		}
	}
}
