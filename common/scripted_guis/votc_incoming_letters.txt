# VOTC-194 §19: the master letter toggle + unified frequency presets + test
# letter moved out of the retired events .140/.195-.198 into the Incoming
# Letters window (gui/custom_gui/votc_incoming_letters_window.gui).
#
# §11 state-named gating contract (carried over from the events): the loc keys
# name the CURRENT state (opt_letters_on = letters currently ON), while the
# button label is the AVAILABLE ACTION.  So the GUI visible conditions are
# inverted vs the key names - the "Disable" button (key opt_letters_on) is
# visible when the letters are NOT disabled, and the "Enable" button (key
# opt_letters_off) is visible when they are.  Do NOT swap the gating.

# PERSISTENT-SETTINGS contract: the variables written by these sguis
# (votc_ce_initiative_no_letters, votc_incoming_letters_enabled,
# votc_incoming_letters_opt_out, votc_incoming_letters_frequency and the
# per-topic *_frequency variables) are user settings serialized into the
# save game.  NEVER add a `days` argument to any of the
# set_global_variable / set_variable writes below - a timed variable would
# silently expire and reset the player's choices.

# Master toggle - enable branch (mirrors the retired .140 enable branch).
votc_incoming_letters_master_enable_sgui = {
	scope = character
	effect = {
		remove_global_variable = votc_ce_initiative_no_letters
		# Default-on policy: shared with the game-start hook and the enable
		# scripted gui so all three paths set identical defaults.
		votc_incoming_letters_apply_default_settings_effect = yes
	}
}

# Master toggle - disable branch (mirrors the retired .140 disable branch).
votc_incoming_letters_master_disable_sgui = {
	scope = character
	effect = {
		set_global_variable = {
			name = votc_ce_initiative_no_letters
			value = yes
		}
		remove_variable = votc_incoming_letters_enabled
		# Opt-out marker: an explicit "off" must never be silently reverted by
		# the game-start default-on hook (mirrors the old disable decision).
		set_variable = {
			name = votc_incoming_letters_opt_out
			value = 1
		}
	}
}

# VOTC-194 §18/§19: unified letter-frequency presets.  One preset drives BOTH
# letter channels: the NPC-letter global (votc_ce_initiative_letter_freq, read
# by the .5 heartbeat re-arm chain) and the incoming-letter PLAYER variable
# (votc_incoming_letters_frequency - every consumer reads it on the player
# scope, so the old global writes were dead knobs).  Presets write ONLY these
# two storages: the master toggle and the votc_incoming_letters_opt_out marker
# are untouched, so an explicit "off" survives a preset click.
votc_incoming_letters_unified_low_sgui = {
	scope = character
	effect = {
		set_global_variable = {
			name = votc_ce_initiative_letter_freq
			value = 365
		}
		set_variable = {
			name = votc_incoming_letters_frequency
			value = 1
		}
	}
}

# VOTC-194 §30.2.1: the letter parchment was closed.  When the letter that was
# just read is a replyable AI letter (votc_letter_replyable), open the compose
# dialog in REPLY mode: consume the marker, arm votc_letter_replying (the
# compose widget's reply-preview scrollbox and the conditional title read it),
# and hand the sender to message_event.360 as scope:recipient.  The saved
# scopes propagate to the triggered event exactly like the §25.1.1 option
# pattern.  A non-replyable close (fresh player letter, whisper-hijacked
# reply) never satisfies the gate, so this is a no-op there.
votc_letter_parchment_closed_sgui = {
	scope = character
	effect = {
		if = {
			limit = { has_global_variable = votc_letter_replyable }
			remove_global_variable = votc_letter_replyable
			set_global_variable = { name = votc_letter_replying value = yes }
			# VOTC-194 §39: derive the reply sender from the ARTIFACT ITSELF
			# (scope:artifact.creator - the stock artifact->character event target,
			# proven in common/scripted_triggers/00_artifact_triggers.txt).  The
			# app's create_artifact sets creator = the letter's NPC sender, so the
			# binding is intrinsic to THIS letter and can never be stale: a second
			# letter arriving while the first parchment is open cannot redirect
			# the compose dialog.  The old votc_letter_reply_sender global was a
			# single shared slot written by 10 arrival sites and never cleared
			# until compose Send/Close, so a stale value leaked into the next
			# letter's "Reply to Letter from X" title.
			if = {
				limit = { exists = global_var:votc_latest_letter }
				global_var:votc_latest_letter = {
					save_scope_as = votc_letter_reply_target
				}
			}
			# Fallback for a letter whose artifact carries no creator (older
			# saves / a hand-fired test letter): keep the global, but only when
			# it still resolves.  It is cleared on compose Send/Close (§39.2), so
			# it can never outlive the letter it was written for.
			if = {
				limit = {
					NOT = { exists = scope:votc_letter_reply_target }
					exists = global_var:votc_letter_reply_sender
				}
				global_var:votc_letter_reply_sender = { save_scope_as = votc_letter_reply_target }
			}
			if = {
				limit = { exists = scope:votc_letter_reply_target }
				scope:votc_letter_reply_target = { save_scope_as = recipient }
				# Overwrite the shared global with THIS letter's sender so the
				# compose title (votc_letter_reply_title reads the global) is
				# correct even if a previous letter left a stale value.  The
				# global is removed on compose Send/Close (message_event.360),
				# so it can never outlive this letter.
				set_global_variable = { name = votc_letter_reply_sender value = scope:votc_letter_reply_target }
			}
			save_scope_as = actor
			trigger_event = message_event.360
		}
	}
}

# VOTC-194 §37.2.4 (mod half): display trigger for incoming letters.  The
# arrival events .0101-.0108 are hidden, so their event-window widgets never
# instantiate and nothing opens the parchment when the app's run file sets
# votc_latest_letter.  Contract: the app's incoming run file sets
# votc_incoming_letter_popup (and mirrors the slot's sender into
# message_second_scope) right before its cleanup lines.  The _popup state in
# gui/custom_gui/letters_runner.gui (recreated every ~2s) fires this sgui,
# which consumes the marker and triggers the stock non-hidden
# message_event.362 - whose widget opens gui/custom_gui/votc_letter.gui and
# whose immediate arms votc_letter_replyable + re-derives the reply sender
# from message_second_scope.  Inert until the app writes the marker;
# stock/npcltr letters never set it, so there is no double-open.  Known edge:
# two incoming run files landing in the same ~2s window share one marker, so
# only the newest artifact is displayed (the other stays in inventory + the
# Letters tab).
votc_incoming_letter_display_sgui = {
	scope = character
	effect = {
		if = {
			limit = { has_global_variable = votc_incoming_letter_popup }
			# Consume the marker so the _popup watcher state stops.
			remove_global_variable = votc_incoming_letter_popup
			trigger_event = message_event.362
		}
	}
}
