# Triggers for the Scene Selection System
# Each trigger checks if a specific scene is valid for the current context

# --- Generic Scenes ---

is_valid_scene_throneroom = {
	AND = {
		scope:actor.location = scope:recipient.location
		scope:actor = { is_imprisoned = no }
		scope:recipient = { is_imprisoned = no }
		NOT = { scope:actor.culture = { has_coa_gfx = chinese_group_coa_gfx } }
		OR = {
			scope:recipient = { is_courtier_of = scope:actor.location.province_owner }
			scope:recipient = scope:actor.location.province_owner
		}
	}
}

is_valid_scene_garden = {
	AND = {
		scope:actor.location = scope:recipient.location
		scope:actor = { is_imprisoned = no }
		scope:recipient = { is_imprisoned = no }
		OR = {
			scope:recipient = { is_courtier_of = scope:actor.location.province_owner }
			scope:recipient = scope:actor.location.province_owner
		}
	}
}

is_valid_scene_bedchamber = {
	AND = {
		scope:actor.location = scope:recipient.location
		scope:actor = { is_imprisoned = no }
		scope:recipient = { is_imprisoned = no }
		OR = {
			scope:actor = { primary_partner = scope:recipient }
			scope:actor = { has_relation_lover = scope:recipient }
		}
	}
}

is_valid_scene_tavern = {
	AND = {
		scope:actor = { is_imprisoned = no }
		scope:recipient = { is_imprisoned = no }
		NOT = { scope:actor.location.culture = { has_coa_gfx = chinese_group_coa_gfx } }
		OR = {
			scope:actor.location = { has_building_or_higher = castle_01 }
			scope:actor.location = { has_building_or_higher = city_01 }
			scope:actor.location = { has_building_or_higher = temple_01 }
		}
	}
}

is_valid_scene_market = {
	AND = {
		scope:actor = { is_imprisoned = no }
		scope:recipient = { is_imprisoned = no }
		scope:actor.location = { has_building_or_higher = city_01 }
	}
}

is_valid_scene_dungeon = {
	OR = {
		scope:actor = { is_imprisoned = yes }
		scope:recipient = { is_imprisoned = yes }
	}
}

is_valid_scene_feast = {
	AND = {
		scope:actor = { is_imprisoned = no }
		scope:recipient = { is_imprisoned = no }
		OR = {
			has_character_flag = is_feasting
			scope:recipient = { has_character_flag = is_feasting }
		}
	}
}

is_valid_scene_temple = {
	AND = {
		scope:actor = { is_imprisoned = no }
		scope:recipient = { is_imprisoned = no }
		scope:actor.location = { has_building_or_higher = temple_01 }
		NOT = { is_valid_scene_lingyinsi = yes }
	}
}

# --- Specific / Landmark Scenes ---

is_valid_scene_iron_throne = {
	AND = {
		scope:actor.location = scope:recipient.location
		OR = {
			scope:actor = { has_title = title:e_the_iron_throne }
			scope:recipient = { has_title = title:e_the_iron_throne }
		}
	}
}

is_valid_scene_harem = {
	AND = {
		scope:actor.location = scope:recipient.location
		scope:actor = { is_imprisoned = no }
		scope:recipient = { is_imprisoned = no }
		scope:actor.culture = {
			NOT = { has_coa_gfx = chinese_group_coa_gfx }
		}
		OR = {
			scope:actor = { has_court_position = str_slave_concubine_court_position }
			scope:recipient = { has_court_position = str_slave_concubine_court_position }
			scope:actor = { is_concubine_of = scope:recipient }
			scope:recipient = { is_concubine_of = scope:actor }
		}
	}
}

is_valid_scene_lingyinsi = {
	AND = {
		scope:actor.location = scope:recipient.location
		scope:actor = { is_imprisoned = no }
		scope:recipient = { is_imprisoned = no }
		scope:actor.location = { has_building = lingyin_temple_01 }
	}
}
