talk_interaction = {
	category = interaction_category_friendly
	use_diplomatic_range = yes
	ignores_pending_interaction_block = yes
	interface_priority = 200
	common_interaction = yes
	desc = talk_interaction.desc


	ai_will_do = {
		base = 0
	}

	is_valid = {
		OR = {
			custom_description = {
				text = not_in_same_province
				scope:actor.location = scope:recipient.location
				
			}
			scope:actor = {
				is_available_adult = yes
			}
			
		}
	}
	
	on_accept = {
		populate_valid_scenes_list = yes

		if = {
			limit = {
				OR = {
					scope:talk_scene_1 = yes
					scope:talk_scene_2 = yes
					scope:talk_scene_3 = yes
					scope:talk_scene_4 = yes
					scope:talk_scene_5 = yes
					scope:talk_scene_6 = yes
					scope:talk_scene_7 = yes
					scope:talk_scene_8 = yes
					scope:talk_scene_9 = yes
					scope:talk_scene_10 = yes
				}
			}
			# Set the selected scene from the interaction option
			if = { limit = { scope:talk_scene_1 = yes } set_global_variable = { name = talk_scene value = scope:votc_scene_1 } }
			else_if = { limit = { scope:talk_scene_2 = yes } set_global_variable = { name = talk_scene value = scope:votc_scene_2 } }
			else_if = { limit = { scope:talk_scene_3 = yes } set_global_variable = { name = talk_scene value = scope:votc_scene_3 } }
			else_if = { limit = { scope:talk_scene_4 = yes } set_global_variable = { name = talk_scene value = scope:votc_scene_4 } }
			else_if = { limit = { scope:talk_scene_5 = yes } set_global_variable = { name = talk_scene value = scope:votc_scene_5 } }
			else_if = { limit = { scope:talk_scene_6 = yes } set_global_variable = { name = talk_scene value = scope:votc_scene_6 } }
			else_if = { limit = { scope:talk_scene_7 = yes } set_global_variable = { name = talk_scene value = scope:votc_scene_7 } }
			else_if = { limit = { scope:talk_scene_8 = yes } set_global_variable = { name = talk_scene value = scope:votc_scene_8 } }
			else_if = { limit = { scope:talk_scene_9 = yes } set_global_variable = { name = talk_scene value = scope:votc_scene_9 } }
			else_if = { limit = { scope:talk_scene_10 = yes } set_global_variable = { name = talk_scene value = scope:votc_scene_10 } }

			scope:actor = { 
				trigger_event = talk_event.9001
			}
		}
		else = {
			if = {
				limit = {
					scope:talk_travel = yes
				}
				
				scope:actor = {
					start_travel_plan = {
					destination = scope:recipient.location
					on_arrival_event = talk_event.1001
					}
				}
				
			}
			
		}
	}

	send_option = {
		is_shown = { exists = var:votc_scene_1 }
		flag = talk_scene_1
		localization = VOTC_SCENE_OPTION_1
	}
	send_option = {
		is_shown = { exists = var:votc_scene_2 }
		flag = talk_scene_2
		localization = VOTC_SCENE_OPTION_2
	}
	send_option = {
		is_shown = { exists = var:votc_scene_3 }
		flag = talk_scene_3
		localization = VOTC_SCENE_OPTION_3
	}
	send_option = {
		is_shown = { exists = var:votc_scene_4 }
		flag = talk_scene_4
		localization = VOTC_SCENE_OPTION_4
	}
	send_option = {
		is_shown = { exists = var:votc_scene_5 }
		flag = talk_scene_5
		localization = VOTC_SCENE_OPTION_5
	}
	send_option = {
		is_shown = { exists = var:votc_scene_6 }
		flag = talk_scene_6
		localization = VOTC_SCENE_OPTION_6
	}
	send_option = {
		is_shown = { exists = var:votc_scene_7 }
		flag = talk_scene_7
		localization = VOTC_SCENE_OPTION_7
	}
	send_option = {
		is_shown = { exists = var:votc_scene_8 }
		flag = talk_scene_8
		localization = VOTC_SCENE_OPTION_8
	}
	send_option = {
		is_shown = { exists = var:votc_scene_9 }
		flag = talk_scene_9
		localization = VOTC_SCENE_OPTION_9
	}
	send_option = {
		is_shown = { exists = var:votc_scene_10 }
		flag = talk_scene_10
		localization = VOTC_SCENE_OPTION_10
	}

	send_option = {
		is_shown = {
			NOT = {scope:actor.location = scope:recipient.location}
		}
		flag = talk_travel
	}
	
	
	auto_accept = yes
}
