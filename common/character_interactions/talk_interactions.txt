talk_interaction = {
	category = interaction_category_friendly
	use_diplomatic_range = yes
	ignores_pending_interaction_block = yes
	interface_priority = 200
	common_interaction = yes
	desc = talk_interaction.desc


	ai_will_do = {
		base = 0
	}

	is_valid = {
		OR = {
			custom_description = {
				text = not_in_same_province
				scope:actor.location = scope:recipient.location
				
			}
			scope:actor = {
				is_available_adult = yes
			}
			
		}
	}
	
	on_accept = {
		set_global_variable = {
			name = talk_first_scope
			value = scope:actor
		}
		set_global_variable = {
			name = talk_second_scope
			value = scope:recipient
		}
		if = {
			limit = {
				scope:actor.location = scope:recipient.location
			}
			scope:actor = { 
				execute_console_command = "gui.createwidget gui/custom_gui/scene_picker.gui scene_picker_window"
			}
		}
		else = {
			scope:actor = {
				start_travel_plan = {
					destination = scope:recipient.location
					on_arrival_event = talk_event.1001 # Keep this for now as a fallback or update it later
				}
			}
		}
	}

	send_option = {
		is_shown = {
			NOT = { scope:actor.location = scope:recipient.location }
		}
		flag = talk_travel
	}
	auto_accept = yes
	
}
