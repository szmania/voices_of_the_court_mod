# VOTC AI-initiated conversations (VOTC-182): an AI character at the player's
# location can ask to speak with the player.
#
# Hooked via the official forwarding pattern - a second effect or trigger block
# for the same on_action is silently ignored (see votc_game_start_init_on_actions.txt
# and vanilla common/on_action/_on_actions.info), so the work lives in a mod-owned
# relay on_action instead.
#
# quarterly_playable_pulse runs once per quarter (the most frequent built-in
# playable-character pulse; CK3 has no monthly pulse) for every playable character
# with the character as its scope; is_ai = no narrows it to human players (in
# multiplayer every player gets their own pulse run). Cooldown flags keep the
# cadence from spamming.
quarterly_playable_pulse = {
	on_actions = {
		votc_ai_initiate_conversation_pulse
	}
}

votc_ai_initiate_conversation_pulse = {
	trigger = {
		is_ai = no
		# The cooldown flag is set on the player when a request is created, and on
		# the AI initiator too, so neither side is asked again immediately.
		NOT = { has_character_flag = votc_ai_initiate_cooldown }
	}
	effect = {
		votc_ai_initiate_conversation_effect = yes
	}
}
