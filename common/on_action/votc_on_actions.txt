# VOTC On Actions - Daily tick and clipboard update
# Ensures VOTC:DATE clipboard code is updated daily and on game start

# Pattern: Initialize a hidden self-looping event via on_game_start_after_lobby
# The daily tick event will reschedule itself every day and trigger mcc_event_v2.9999

# Hook into vanilla on_game_start_after_lobby (called after lobby exit)
# Use on_actions to avoid conflicting effect blocks
votc_on_game_start_hook = {
	effect = {
		# Initialize the daily tick loop for each player
		every_player = {
			trigger_event = votc_daily_tick.0001
		}
	}
}

# Merge into vanilla on_action
on_game_start_after_lobby = {
	on_actions = { votc_on_game_start_hook }
}

# Re-arm the daily tick loop for saves created BEFORE this mod was added.
# There is no on_save_game_loaded on_action in vanilla, so we piggyback on the
# existing recurring playable pulses via the on_actions merge pattern. The loop
# only fires when the character is not already armed (see
# votc_daily_tick.0001 immediate), so arming does not duplicate the loop.
quarterly_playable_pulse = {
	on_actions = { votc_quarterly_rearm_hook }
}

votc_quarterly_rearm_hook = {
	effect = {
		if = {
			limit = { NOT = { has_variable = votc_daily_tick_armed } }
			trigger_event = votc_daily_tick.0001
		}
	}
}
