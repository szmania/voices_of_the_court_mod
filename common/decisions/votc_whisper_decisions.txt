#==========================================================================
#  The Whisper Ledger  -  decisions  (VOTC-195)
#  VOTC-CE decision-group entries.  One decision opens the settings hub,
#  one opens the ledger report window, one re-opens the most recent report.
#==========================================================================

votc_whisper_settings_decision = {
	decision_group_type = votc_ce

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}

	sort_order = 89

	desc = votc_whisper_settings_decision_desc
	selection_tooltip = votc_whisper_settings_decision_tooltip

	is_shown = {
		is_ai = no
	}

	is_valid = {
		is_ai = no
	}

	cooldown = { days = 1 }

	effect = {
		if = {
			limit = { NOT = { has_global_variable = votc_whisper_freq } }
			set_global_variable = {
				name = votc_whisper_freq
				value = 182
			}
		}
		trigger_event = votc_whisper.100
	}

	ai_check_interval = 0
	ai_will_do = {
		base = 0
}
