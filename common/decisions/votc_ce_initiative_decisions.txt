#==========================================================================
#  VOTC-194: NPC Initiative - decision
#  A tiny always-available entry point in the Decisions tab that opens the
#  configuration window (votc_ce_initiative.100).
#==========================================================================

votc_ce_initiative_settings_decision = {
	decision_group_type = votc_ce

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}

	sort_order = 90

	desc = votc_ce_initiative_settings_decision_desc
	selection_tooltip = votc_ce_initiative_settings_decision_tooltip

	is_shown = {
		NOT = { is_ai = yes }
	}
	# VOTC-194 §38.2.2: the is_valid block was deleted.  It was 100% redundant
	# (is_shown already hides the decision from AI; ai_check_interval = 0 and
	# ai_will_do = { base = 0 } keep AI from ever taking it; for a human player
	# it always evaluated true) and it rendered a confusing "Is AI" requirement
	# line in the decision tooltip.  Removing it also eliminates the vanilla
	# trigger-localization warning class (is_ai has no 'first_not' perspective in
	# common/trigger_localization/00_debug_triggers.txt) deterministically.
	effect = {
		# Make sure the settings exist before the window reads them.
		if = {
			limit = { NOT = { has_global_variable = votc_ce_initiative_freq } }
			set_global_variable = {
				name = votc_ce_initiative_freq
				value = 182
			}
		}
		if = {
			limit = { NOT = { has_global_variable = votc_ce_initiative_chance } }
			set_global_variable = {
				name = votc_ce_initiative_chance
				value = 15
			}
		}
		if = {
			limit = { NOT = { has_global_variable = votc_ce_initiative_letter_freq } }
			set_global_variable = {
				name = votc_ce_initiative_letter_freq
				value = 182
			}
		}
		trigger_event = votc_ce_initiative.100
	}

	ai_check_interval = 0
	ai_will_do = {
		base = 0
	}
}
